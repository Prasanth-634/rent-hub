# Database Schema

users(id UUID PK, email unique, password_hash, full_name, role, is_active, created_at, updated_at)
organizations(id UUID PK, name, type, created_at)
organization_users(organization_id FK, user_id FK, role)
subscription_plans(id UUID PK, name, price_minor_units, currency, monthly_request_limit, active)
subscriptions(id UUID PK, organization_id FK, plan_id FK, status, starts_at, expires_at, provider_customer_id, provider_subscription_id)
payments(id UUID PK, organization_id FK, provider, provider_payment_id unique, amount_minor_units, currency, status, raw_event_hash, paid_at, created_at)
api_keys(id UUID PK, organization_id FK, key_prefix, secret_hash, status, created_at, last_used_at, expires_at)
tenants(id UUID PK, user_id nullable FK, full_name, email, phone, created_at)
properties(id UUID PK, organization_id FK, address_line1, city, state, postal_code, country, created_at)
leases(id UUID PK, property_id FK, tenant_id FK, monthly_rent_minor_units, currency, due_day, start_date, end_date, status)
verification_requests(id UUID PK, external_id unique, requester_organization_id FK, tenant_id FK, lease_id FK, period_start, period_end, status, created_at, completed_at)
consents(id UUID PK, verification_id FK, tenant_id FK, status, scope, granted_at, revoked_at, expires_at)
transactions(id UUID PK, verification_id FK, external_transaction_id, transaction_date, amount_minor_units, currency, description, payer, payee, is_rent_predicted, rent_probability, anomaly_score, anomaly_flag)
verification_results(id UUID PK, verification_id FK unique, months_expected, months_matched, on_time_count, late_count, partial_count, missed_count, duplicate_count, confidence_level, status, explanation_json, finalized_at)
audit_logs(id UUID PK, actor_user_id, action, resource_type, resource_id, metadata_json, created_at)
notifications(id UUID PK, recipient, type, status, provider_message_id, created_at)
webhook_events(id UUID PK, provider, event_id unique, event_type, processed_at)
